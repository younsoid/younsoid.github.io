                              NGS Report 정제 - CNV, Fusion, MSI, TMB #

                              #2026-07-06

                              1. 작업 개요 #

                              기존 분석에서 분석 대상자 368명 중 FMI 데이터가 있는 27명에 대해서 데이터 추출 작업을 수행해줬다. 이번 분석에서는 CNV, Fusion, MSI, TMB 데이터가 있는 대상자에 대해서 데이터 추출 작업을 진행할 예정이다.

                              #

                              2. 작업 내용 #

                              import os
import pandas as pd

data_tool_01 = pd.read_excel(r"C:\Users\User\Documents\yshid\2607\data\20260527_Data tool_20260527104012951.xlsx", sheet_name=0, engine="openpyxl")
data_tool_02 = pd.read_excel(r"C:\Users\User\Documents\yshid\2607\data\20260527_Data tool_20260527104012951.xlsx", sheet_name=6, engine="openpyxl")
data_tool_03 = pd.read_excel(r"C:\Users\User\Documents\yshid\2607\data\20260527_Data tool_20260527104012951.xlsx", sheet_name=1, engine="openpyxl")
data_tool_04 = pd.read_excel(r"C:\Users\User\Documents\yshid\2607\data\20260527_Data tool_20260527104012951.xlsx", sheet_name=2, engine="openpyxl")

data_tool_01 = data_tool_01[['SUBJ_ID', 'CRF_REPEATKEY', 'QC_PASS_YN', 'TUMR_PRIT_RATE', 'TMB_TOTL_CNT', 'MSI', 'BRND_CD', 'BRND_NM_VER']]
data_tool_02 = data_tool_02[['SUBJ_ID', 'CRF_REPEATKEY', 'SNV_VRNT_YN', 'BRND_CD_REL_3', 'BRND_NM_VER_REL_3', 'ITEMGROUP_REPEATKEY_1', 'SNV_GENE_CMNT', 'DNA_CHNG', '3_LETR_AA_CHNG', '1_LETR_AA_CHNG', 'VAF', 'TIRE_CD3']]
data_tool_03 = data_tool_03[['SUBJ_ID', 'CRF_REPEATKEY', 'BRND_CD_REL_1', 'BRND_NM_VER_REL_1', 'ITEMGROUP_REPEATKEY_1', 'GENE_CMNT', 'GENE_CHNG_SITE_NM', 'GENE_CHNG_SITE_CHR', 'CHRM_SOME_CMNT', 'STRT_LOCA_CMNT', 'END_LOCA_CMNT', 'COYP_NO_ATRT_TYPE', 'ECT', 'FCCN_CNT', 'CN_CNT', 'ADJV_CN_CNT', 'TIRE_CD', 'EBDN_LEVL_CD']]
data_tool_04 = data_tool_04[['SUBJ_ID', 'CRF_REPEATKEY', 'VRNT_CD', 'BRND_CD_REL_2', 'BRND_NM_VER_REL_2', 'ITEMGROUP_REPEATKEY_1', 'FUSI_GENE', 'GENE1_CMNT', 'GENE1_BREK_PONT', 'GENE2_CMNT', 'GENE2_BREK_PONT', 'FUSI_SPPR_READ_CNT', 'DTCT_MTHD_CD', 'TIRE_CD2', 'EBDN_LEVL_CD2']]

with open(r"C:\Users\User\Documents\yshid\2607\data\add_total_sid_0702.txt", "r") as f:
    add_sid = [line.strip() for line in f if line.strip()]
    
len(add_sid) # 368

                              CNV, Fusion, MSI, TMB 데이터는 eCRF의 Data Tool 파일에 존재하는데, sheet 0에 MSI와 TMB, sheet 1에 CNV 정보, sheet 2에 Fusion 정보가 있다.

                              DATA_DIR = r"C:\Users\User\Documents\yshid\2607\data"

MSI_MAP  = {"1": "MSS", "2": "MSI-L", "3": "MSI-H", "4": ""}                       # 4=Other→빈칸
COYP_MAP = {"1": "Gain", "2": "Loss", "3": "Amplification", "4": "Deletion", "5": ""}  # 5=Other→빈칸

add_up = {s.strip().upper() for s in add_sid}
seq_ids = set(data_tool_01["SUBJ_ID"].dropna().astype(str).str.upper())
missing = sorted(add_up - seq_ids)
print(len(add_up), len(add_up & seq_ids), len(missing)) # 368 329 39

                              MSI 컬럼은 1이 MSS, 2가 MSI-L, 3이 MSI-H, 4가 Other이다. Copy Number Type은 1이 Gain, 2가 Loss, 3이 Amplification, 4가 Deletion, 5가 Other이다.

                              기존 분석 대상자 368명 중 NGS Report가 있는 대상자를 확인했을때 329명은 정보가 있고 39명은 존재하지 않았다. 이에 329명에 대해서 CNV, Fusion, MSI, TMB 데이터를 추출해줬다.

                              seq = data_tool_01[data_tool_01["SUBJ_ID"].astype(str).str.upper().isin(add_up)].reset_index(drop=True)

msi_tmb = pd.DataFrame({
    "SUBJ_ID":        seq["SUBJ_ID"],
    "TUMR_PRIT_RATE": seq["TUMR_PRIT_RATE"].map(_num),
    "TMB_TOTL_CNT":   seq["TMB_TOTL_CNT"].map(_num),
    "MSI":            seq["MSI"].map(lambda v: MSI_MAP.get(_code(v), "")),
    "QC_PASS":        seq["QC_PASS_YN"].map(lambda v: _code(v) or ""), 
})
tmb = msi_tmb[["SUBJ_ID", "TMB_TOTL_CNT", "QC_PASS"]].copy()
print(len(msi_tmb), len(tmb), msi_tmb["SUBJ_ID"].nunique()) # 338, 338, 329

                              먼저 MSI와 TMB 정보를 sheet 0에서 가져왔다. 추출 결과 데이터 수는 MSI 338개, TMB 338개였고 대상자는 329명에 대해 모두 존재했다.

                              cnv_src = data_tool_03[
    data_tool_03["SUBJ_ID"].astype(str).str.upper().isin(add_up)
    & data_tool_03["GENE_CMNT"].notna()
].reset_index(drop=True)

cnv = pd.DataFrame({
    "SUBJ_ID":           cnv_src["SUBJ_ID"],
    "GENE_CMNT":         cnv_src["GENE_CMNT"],
    "COYP_NO_ATRT_TYPE": cnv_src["COYP_NO_ATRT_TYPE"].map(lambda v: COYP_MAP.get(_code(v), "")),
})
print(len(cnv), cnv["SUBJ_ID"].nunique()) # 1254 199
print(cnv["COYP_NO_ATRT_TYPE"].value_counts(dropna=False).to_dict()) 
# {'Amplification': 528, 'Loss': 386, 'Gain': 175, 'Deletion': 108, '': 57}

                              다음으로 CNV 데이터를 sheet 2에서 가져왔다. 추출 결과 데이터 수는 1,254개였고 대상자 199명에 대해 존재했다. COYP_NO_ATRT_TYPE 컬럼의 값은 Amplification 528개, Loss 386개, Gain 175개, Deletion 108개, Other이 57개였다.

                              fus_src = data_tool_04[
    data_tool_04["SUBJ_ID"].astype(str).str.upper().isin(add_up)
    & data_tool_04["GENE1_CMNT"].notna()
].reset_index(drop=True)

lc = fus_src["GENE1_BREK_PONT"].map(_split_bp)
rc = fus_src["GENE2_BREK_PONT"].map(_split_bp)
fusion = pd.DataFrame({
    "sample_id":       fus_src["SUBJ_ID"],
    "left_gene_name":  fus_src["GENE1_CMNT"],
    "left_gene_chr":   [x[0] for x in lc],
    "left_hg38_pos":   [x[1] for x in lc],
    "right_gene_name": fus_src["GENE2_CMNT"],
    "right_gene_chr":  [x[0] for x in rc],
    "right_hg38_pos":  [x[1] for x in rc],
})
print(len(fusion), fusion["sample_id"].nunique()) # 112 38

                              마지막으로 Fusion 데이터를 sheet 3에서 가져왔다. 추출 결과 데이터 수는 112개였고 대상자 38명에 대해 존재했다.

                              msi_tmb.to_csv(os.path.join(DATA_DIR, "326명_대상자_MSI_TMB.csv"), index=False, encoding="utf-8-sig")
tmb.to_csv(os.path.join(DATA_DIR, "326명_대상자_TMB.csv"),     index=False, encoding="utf-8-sig")
cnv.to_csv(os.path.join(DATA_DIR, "326명_대상자_CNV.csv"),     index=False, encoding="utf-8-sig")
fusion.to_csv(os.path.join(DATA_DIR, "326명_대상자_Fusion.csv"),  index=False, encoding="utf-8-sig")

                              추출한 데이터프레임을 csv 파일로 저장해줬다.
